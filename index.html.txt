    <script>
        const canvas = document.getElementById('signatureCanvas');
        const ctx = canvas.getContext('2d');
        let isDrawing = false;
        
        let attachedFiles = [];

        function initCanvas() {
            ctx.strokeStyle = "#000000";
            ctx.lineWidth = 2;
            ctx.lineCap = "round";

            canvas.addEventListener('mousedown', startDrawing);
            canvas.addEventListener('mousemove', draw);
            canvas.addEventListener('mouseup', stopDrawing);
            canvas.addEventListener('mouseleave', stopDrawing);

            canvas.addEventListener('touchstart', (e) => {
                const touch = e.touches[0];
                const rect = canvas.getBoundingClientRect();
                isDrawing = true;
                ctx.beginPath();
                ctx.moveTo(touch.clientX - rect.left, touch.clientY - rect.top);
                e.preventDefault();
            });
            canvas.addEventListener('touchmove', (e) => {
                if (!isDrawing) return;
                const touch = e.touches[0];
                const rect = canvas.getBoundingClientRect();
                ctx.lineTo(touch.clientX - rect.left, touch.clientY - rect.top);
                ctx.stroke();
                e.preventDefault();
            });
            canvas.addEventListener('touchend', stopDrawing);
        }

        function startDrawing(e) {
            isDrawing = true;
            ctx.beginPath();
            ctx.moveTo(e.offsetX, e.offsetY);
        }

        function draw(e) {
            if (!isDrawing) return;
            ctx.lineTo(e.offsetX, e.offsetY);
            ctx.stroke();
        }

        function stopDrawing() {
            if (isDrawing) {
                isDrawing = false;
                syncPDF();
            }
        }

        function clearSignature() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            syncPDF();
        }

        function handleFileUpload(event) {
            const files = event.target.files;
            if (files && files.length > 0) {
                Array.from(files).forEach(file => {
                    const reader = new FileReader();
                    reader.onload = function(e) {
                        attachedFiles.push({
                            file: file,
                            bytes: e.target.result,
                            type: file.type.includes('pdf') ? 'pdf' : 'image',
                            name: file.name
                        });
                        updateFilesUI();
                        syncPDF();
                    };
                    reader.readAsArrayBuffer(file);
                });
            }
        }

        function removeFile(index) {
            attachedFiles.splice(index, 1);
            updateFilesUI();
            syncPDF();
        }

        function updateFilesUI() {
            const container = document.getElementById('fileListContainer');
            const listItems = document.getElementById('filesListItems');
            listItems.innerHTML = '';

            if (attachedFiles.length === 0) {
                container.classList.add('hidden');
                return;
            }

            container.classList.remove('hidden');
            attachedFiles.forEach((item, index) => {
                const isPdf = item.type === 'pdf';
                const iconClass = isPdf ? 'fa-solid fa-file-pdf text-rose-600' : 'fa-solid fa-file-image text-emerald-600';
                
                const div = document.createElement('div');
                div.className = "flex items-center justify-between text-xs bg-white p-2 rounded border border-slate-200 shadow-sm";
                div.innerHTML = `
                    <div class="flex items-center gap-2 truncate">
                        <i class="${iconClass} text-sm"></i>
                        <span class="font-bold text-slate-700 truncate">${item.name}</span>
                    </div>
                    <button type="button" onclick="removeFile(${index})" class="text-rose-600 hover:text-rose-800 font-semibold px-2"><i class="fa-solid fa-trash"></i></button>
                `;
                listItems.appendChild(div);
            });
        }

        function clearAllFiles() {
            attachedFiles = [];
            document.getElementById('docArquivos').value = '';
            updateFilesUI();
            syncPDF();
        }

        function validarCPF(cpf) {
            cpf = cpf.replace(/[^\d]+/g, '');
            if (cpf.length !== 11 || /^(\d)\1{10}$/.test(cpf)) return false;
            let soma = 0;
            for (let i = 1; i <= 9; i++) soma += parseInt(cpf.substring(i - 1, i)) * (11 - i);
            let resto = (soma * 10) % 11;
            if ((resto === 10) || (resto === 11)) resto = 0;
            if (resto !== parseInt(cpf.substring(9, 10))) return false;
            soma = 0;
            for (let i = 1; i <= 10; i++) soma += parseInt(cpf.substring(i - 1, i)) * (12 - i);
            resto = (soma * 10) % 11;
            if ((resto === 10) || (resto === 11)) resto = 0;
            if (resto !== parseInt(cpf.substring(10, 11))) return false;
            return true;
        }

        function formatCPF(input) {
            let value = input.value.replace(/\D/g, "");
            value = value.replace(/(\d{3})(\d)/, "$1.$2");
            value = value.replace(/(\d{3})(\d)/, "$1.$2");
            value = value.replace(/(\d{3})(\d{1,2})$/, "$1-$2");
            input.value = value;

            const errorSpan = document.getElementById('cpfError');
            if (value.length === 14) {
                if (!validarCPF(value)) {
                    errorSpan.classList.remove('hidden');
                    input.classList.add('border-rose-500');
                } else {
                    errorSpan.classList.add('hidden');
                    input.classList.remove('border-rose-500');
                }
            } else {
                errorSpan.classList.add('hidden');
                input.classList.remove('border-rose-500');
            }
        }

        function onSaidaChange() {
            const dataSaida = document.getElementById('dataSaida').value;
            const dataRetornoInput = document.getElementById('dataRetorno');
            if (dataSaida) {
                dataRetornoInput.min = dataSaida;
                if (!dataRetornoInput.value || dataRetornoInput.value < dataSaida) {
                    dataRetornoInput.value = dataSaida;
                }
            }
        }

        function onModalidadeChange() {
            const mod = document.querySelector('input[name="modalidade"]:checked')?.value;
            const valorInput = document.getElementById('valorUnitario');

            if (mod === 'Estadual') {
                valorInput.value = '220.00';
                valorInput.readOnly = true;
                valorInput.classList.add('bg-slate-100');
            } else if (mod === 'Nacional') {
                valorInput.value = '450.00';
                valorInput.readOnly = true;
                valorInput.classList.add('bg-slate-100');
            } else if (mod === 'Internacional') {
                valorInput.readOnly = false;
                valorInput.classList.remove('bg-slate-100');
            }
            calculateTotal();
            syncPDF();
        }

        function calculateDiarias() {
            const d1 = document.getElementById('dataSaida').value;
            const d2 = document.getElementById('dataRetorno').value;
            const errorSpan = document.getElementById('retornoError');
            const retornoInput = document.getElementById('dataRetorno');

            if (d1 && d2) {
                if (d2 < d1) {
                    errorSpan.classList.remove('hidden');
                    retornoInput.classList.add('border-rose-500');
                    document.getElementById('qtdDiarias').value = '';
                    document.getElementById('valorTotal').value = 'R$ 0,00';
                    syncPDF();
                    return;
                } else {
                    errorSpan.classList.add('hidden');
                    retornoInput.classList.remove('border-rose-500');
                }

                const date1 = new Date(d1 + 'T00:00:00');
                const date2 = new Date(d2 + 'T00:00:00');
                const diffTime = date2 - date1;
                const diffDays = Math.round(diffTime / (1000 * 60 * 60 * 24));

                if (diffDays >= 0) {
                    document.getElementById('qtdDiarias').value = diffDays + 1;
                }
            } else {
                errorSpan.classList.add('hidden');
                retornoInput.classList.remove('border-rose-500');
            }
            calculateTotal();
        }

        function calculateTotal() {
            const qtd = parseFloat(document.getElementById('qtdDiarias').value) || 0;
            const unit = parseFloat(document.getElementById('valorUnitario').value) || 0;
            const total = qtd * unit;

            document.getElementById('valorTotal').value = total.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' });
        }

        function formatDateBR(dateStr) {
            if (!dateStr) return "-";
            const parts = dateStr.split('-');
            if (parts.length === 3) return `${parts[2]}/${parts[1]}/${parts[0]}`;
            return dateStr;
        }

        async function generatePDF() {
            const cpfVal = document.getElementById('cpf').value;
            if (!validarCPF(cpfVal)) {
                alert("Atenção: O CPF inserido é inválido. Por favor, insira um CPF real para continuar.");
                document.getElementById('cpf').focus();
                return;
            }

            const d1 = document.getElementById('dataSaida').value;
            const d2 = document.getElementById('dataRetorno').value;
            if (d1 && d2 && d2 < d1) {
                alert("Atenção: A data de retorno não pode ser anterior à data de saída.");
                document.getElementById('dataRetorno').focus();
                return;
            }

            // Exibe a tela de carregamento
            const loadingOverlay = document.getElementById('loadingOverlay');
            loadingOverlay.classList.remove('hidden');

            syncPDF();

            let modalWasHidden = false;
            const modal = document.getElementById('previewModal');
            const targetContainer = document.getElementById('render-target-container');

            if (modal.classList.contains('hidden')) {
                modalWasHidden = true;
                targetContainer.innerHTML = '';
                const sourceHtml = document.getElementById('pdf-source-html').children[0];
                targetContainer.appendChild(sourceHtml.cloneNode(true));
                modal.classList.remove('hidden');
            }

            await new Promise(resolve => setTimeout(resolve, 200));

            const elementToCapture = targetContainer.children[0];
            const nomeBeneficiario = document.getElementById('nome').value.trim().replaceAll(' ', '_') || 'BENEFICIARIO';
            const filename = `PLANO_DE_VIAGEM_${nomeBeneficiario}.pdf`;

            try {
                const canvasRendered = await html2canvas(elementToCapture, {
                    scale: 2,
                    useCORS: true,
                    logging: false,
                    backgroundColor: '#ffffff'
                });

                if (modalWasHidden) {
                    modal.classList.add('hidden');
                }

                const imgData = canvasRendered.toDataURL('image/jpeg', 0.98);

                const { jsPDF } = window.jspdf;
                const pdfDocJs = new jsPDF('landscape', 'mm', 'a4');
                pdfDocJs.addImage(imgData, 'JPEG', 5, 5, 287, 200);

                const pdfBaseArrayBuffer = pdfDocJs.output('arraybuffer');

                if (attachedFiles.length === 0) {
                    const blob = new Blob([pdfBaseArrayBuffer], { type: 'application/pdf' });
                    downloadBlob(blob, filename);
                    loadingOverlay.classList.add('hidden');
                    return;
                }

                const { PDFDocument } = PDFLib;
                const basePdfDoc = await PDFDocument.load(pdfBaseArrayBuffer);

                for (let item of attachedFiles) {
                    if (item.type === 'pdf') {
                        const attachmentPdfDoc = await PDFDocument.load(item.bytes);
                        const copiedPages = await basePdfDoc.copyPages(attachmentPdfDoc, attachmentPdfDoc.getPageIndices());
                        copiedPages.forEach(page => basePdfDoc.addPage(page));
                    } else if (item.type === 'image') {
                        const newPage = basePdfDoc.addPage([841.89, 595.28]);
                        let embeddedImg;

                        if (item.file.type.includes('png')) {
                            embeddedImg = await basePdfDoc.embedPng(item.bytes);
                        } else {
                            embeddedImg = await basePdfDoc.embedJpg(item.bytes);
                        }

                        const { width, height } = embeddedImg.scale(1);
                        const maxWidth = 800;
                        const maxHeight = 500;
                        const scaleFactor = Math.min(maxWidth / width, maxHeight / height, 1);

                        const finalWidth = width * scaleFactor;
                        const finalHeight = height * scaleFactor;

                        newPage.drawImage(embeddedImg, {
                            x: (841.89 - finalWidth) / 2,
                            y: (595.28 - finalHeight) / 2,
                            width: finalWidth,
                            height: finalHeight,
                        });
                    }
                }

                const mergedPdfBytes = await basePdfDoc.save();
                const finalBlob = new Blob([mergedPdfBytes], { type: 'application/pdf' });
                downloadBlob(finalBlob, filename);

            } catch (err) {
                console.error('Erro na geração do PDF:', err);
                if (modalWasHidden) modal.classList.add('hidden');
                alert('Ocorreu um erro ao gerar o documento. Tente novamente.');
            } finally {
                loadingOverlay.classList.add('hidden');
            }
        }

        function downloadBlob(blob, filename) {
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = filename;
            link.click();
            URL.revokeObjectURL(link.href);
        }
